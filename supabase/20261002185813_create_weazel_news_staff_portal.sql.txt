/*
# Weazel News staff portal foundation

1. New Tables
- `profiles`: one editable staff profile per authenticated employee, with rank and department data.
- `rank_catalog`: the 15-rank Weazel News ladder and promotion requirements.
- `promotion_requests`: employee requests for the next rank and their review state.
- `notifications`: private in-app notifications for each employee.
- `news_articles`: employee-owned drafts and published newsroom materials.
- `editorial_events`: shared newsroom schedule entries.

2. Authentication and ownership
- Every profile is linked to `auth.users` and is created automatically after signup.
- Employee-owned records default their owner from `auth.uid()`.
- Users can edit only their own personal content; rank and review fields are never client-writable.

3. Security
- Row level security is enabled on every new table.
- Policies are separated by CRUD operation.
- Senior staff can review promotion requests through a server-side function that checks the caller's current rank.
- Rank changes create a private notification for the employee in the same server-side operation.

4. Important notes
- The initial signup profile starts at Intern / rank 1.
- Ranks 10 and above can review requests.
- The review function accepts only the request id and decision; it derives the acting staff member from the authenticated session.
*/

CREATE TABLE IF NOT EXISTS public.profiles (
  id uuid PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  full_name text NOT NULL DEFAULT 'Новий співробітник',
  employee_number text NOT NULL DEFAULT 'PENDING',
  rank_number integer NOT NULL DEFAULT 1 CHECK (rank_number BETWEEN 1 AND 15),
  rank_name text NOT NULL DEFAULT 'Intern',
  department text NOT NULL DEFAULT 'Редакція',
  bio text NOT NULL DEFAULT '',
  avatar_url text,
  settings jsonb NOT NULL DEFAULT '{}'::jsonb,
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE IF NOT EXISTS public.rank_catalog (
  rank_number integer PRIMARY KEY CHECK (rank_number BETWEEN 1 AND 15),
  rank_name text UNIQUE NOT NULL,
  requirements text NOT NULL DEFAULT '',
  created_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE IF NOT EXISTS public.promotion_requests (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  employee_id uuid NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
  requested_rank integer NOT NULL CHECK (requested_rank BETWEEN 2 AND 15),
  reason text NOT NULL DEFAULT '',
  status text NOT NULL DEFAULT 'pending' CHECK (status IN ('pending', 'approved', 'rejected')),
  reviewed_by uuid REFERENCES public.profiles(id),
  reviewed_at timestamptz,
  reviewer_note text NOT NULL DEFAULT '',
  created_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE IF NOT EXISTS public.notifications (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  recipient_id uuid NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
  notification_type text NOT NULL DEFAULT 'system',
  title text NOT NULL,
  body text NOT NULL,
  read_at timestamptz,
  created_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE IF NOT EXISTS public.news_articles (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  author_id uuid NOT NULL DEFAULT auth.uid() REFERENCES public.profiles(id) ON DELETE CASCADE,
  title text NOT NULL,
  summary text NOT NULL DEFAULT '',
  status text NOT NULL DEFAULT 'draft' CHECK (status IN ('draft', 'review', 'published')),
  category text NOT NULL DEFAULT 'Загальне',
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE IF NOT EXISTS public.editorial_events (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  created_by uuid NOT NULL DEFAULT auth.uid() REFERENCES public.profiles(id) ON DELETE CASCADE,
  title text NOT NULL,
  event_time timestamptz NOT NULL,
  location text NOT NULL DEFAULT 'Редакція',
  event_type text NOT NULL DEFAULT 'briefing',
  created_at timestamptz NOT NULL DEFAULT now()
);

CREATE OR REPLACE FUNCTION public.handle_new_weazel_profile()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER SET search_path = public
AS $$
BEGIN
  INSERT INTO public.profiles (id, full_name, employee_number)
  VALUES (
    NEW.id,
    COALESCE(NULLIF(NEW.raw_user_meta_data->>'full_name', ''), 'Новий співробітник'),
    COALESCE(NULLIF(NEW.raw_user_meta_data->>'employee_number', ''), 'WN-' || upper(substr(replace(NEW.id::text, '-', ''), 1, 6)))
  )
  ON CONFLICT (id) DO NOTHING;
  RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS on_auth_user_created_weazel ON auth.users;
CREATE TRIGGER on_auth_user_created_weazel
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE FUNCTION public.handle_new_weazel_profile();

CREATE OR REPLACE FUNCTION public.review_weazel_promotion(
  p_request_id uuid,
  p_approved boolean,
  p_note text DEFAULT ''
)
RETURNS public.promotion_requests
LANGUAGE plpgsql
SECURITY DEFINER SET search_path = public
AS $$
DECLARE
  v_actor public.profiles;
  v_request public.promotion_requests;
  v_rank public.rank_catalog;
BEGIN
  SELECT * INTO v_actor FROM public.profiles WHERE id = auth.uid();
  IF v_actor.id IS NULL OR v_actor.rank_number < 10 THEN
    RAISE EXCEPTION 'Not authorized';
  END IF;

  SELECT * INTO v_request FROM public.promotion_requests
  WHERE id = p_request_id AND status = 'pending'
  FOR UPDATE;
  IF v_request.id IS NULL THEN
    RAISE EXCEPTION 'Request is unavailable';
  END IF;

  IF p_approved AND v_request.requested_rank > v_actor.rank_number THEN
    RAISE EXCEPTION 'Cannot grant a rank above your own';
  END IF;

  SELECT * INTO v_rank FROM public.rank_catalog WHERE rank_number = v_request.requested_rank;
  UPDATE public.promotion_requests
  SET status = CASE WHEN p_approved THEN 'approved' ELSE 'rejected' END,
      reviewed_by = auth.uid(), reviewed_at = now(), reviewer_note = left(coalesce(p_note, ''), 500)
  WHERE id = v_request.id
  RETURNING * INTO v_request;

  IF p_approved THEN
    UPDATE public.profiles
    SET rank_number = v_rank.rank_number, rank_name = v_rank.rank_name, updated_at = now()
    WHERE id = v_request.employee_id;
    INSERT INTO public.notifications (recipient_id, notification_type, title, body)
    VALUES (v_request.employee_id, 'promotion_approved', 'Заявку на підвищення схвалено', 'Вітаємо! Ваш новий ранг: ' || v_rank.rank_name || '.');
  ELSE
    INSERT INTO public.notifications (recipient_id, notification_type, title, body)
    VALUES (v_request.employee_id, 'promotion_rejected', 'Заявку на підвищення розглянуто', 'Заявку відхилено. Коментар старшого складу: ' || left(coalesce(p_note, 'Без коментаря'), 500));
  END IF;

  RETURN v_request;
END;
$$;

DROP POLICY IF EXISTS "employees cannot delete profiles" ON public.profiles;

DROP POLICY IF EXISTS "employees cannot delete requests" ON public.promotion_requests;
